  mutate(percent = n / sum(n), year = "1801")

civilstand_percent_1834 <- Census1834 %>%
  filter(!is.na(civilstand)) %>%
  count(civilstand) %>%
  mutate(percent = n / sum(n), year = "1834")

civilstand_percent_1860 <- Census1860 %>%
  filter(!is.na(civilstand)) %>%
  count(civilstand) %>%
  mutate(percent = n / sum(n), year = "1860")

#Combining and plotting the censuses across years 
civilstand_all <- bind_rows(
  civilstand_percent_1801,
  civilstand_percent_1834,
  civilstand_percent_1860) %>%
  mutate(
    civilstand = factor(civilstand, levels = c("ugift", "gift", "enke", "skilt")),
    year = factor(year, levels = c("1801", "1834", "1860"))) %>%
  filter(!is.na(civilstand))

Comparisons <- ggplot(civilstand_all,
  aes(x = civilstand, y = percent, fill = year)) +
  geom_col(position = position_dodge(width = 0.9)) +
  geom_text(
    aes(label = paste0(round(percent * 100, 2), "%")),
    position = position_dodge(width = 0.9),
    vjust = -0.3,
    size = 3,
    fontface = "bold") +
  scale_y_continuous(labels = function(x) paste0(round(x * 100, 2), "%")) +
  expand_limits(y = max(civilstand_all$percent) + 0.05) +
  labs(
    title = "Civil Status Distribution (1801–1860)",
    x = "Civil status",
    y = "Percentage",
    fill = "Year") +
  theme_minimal()
